import json,os,platform,secrets,subprocess,sys
from pathlib import Path
base=Path('/tmp/pkg-runtime-final')
name='pkg-tap-credential-proof-'+secrets.token_hex(8)
sentinel=Path.home()/'.config'/name;sentinel.parent.mkdir(parents=True,exist_ok=True)
secret=secrets.token_hex(32);sentinel.write_text(secret);sentinel.chmod(0o644)
assert sentinel.read_text()==secret
case='credential-sentinel';folder=base/'fixtures/taps'/case/'Casks';folder.mkdir(parents=True,exist_ok=True)
(folder/'pubtap-credential.rb').write_text('''path = PATH
read_denied = begin
  File.binread(path)
  false
rescue Errno::EACCES, Errno::EPERM, Errno::ENOENT
  true
end
write_denied = begin
  File.open(path, "a") { |file| file.write("WRITTEN") }
  false
rescue Errno::EACCES, Errno::EPERM, Errno::ENOENT
  true
end
cask "pubtap-credential" do
  version "1.0.0"
  sha256 "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"
  url "https://example.com/pubtap-credential.tar.gz"
  name "Credential access proof"
  desc "read_denied=#{read_denied};write_denied=#{write_denied}"
  homepage "https://example.com"
  binary "tool"
end
'''.replace('PATH',json.dumps(str(sentinel))))
matrix=base/'public-tap-check/matrix.json';d=json.loads(matrix.read_text());d['source_id'][case]='pubtap-fixtures/credential-sentinel';matrix.write_text(json.dumps(d))
try:
 r=subprocess.run([sys.executable,str(base/'runtime-smoke.py'),str(base),case],capture_output=True,text=True,timeout=300)
 assert r.returncode==0,r.stderr[-1000:]
 system='aarch64-darwin' if platform.system()=='Darwin' else 'x86_64-linux'
 capture=json.loads((base/f'result-{system}-{case}.json').read_text())
 record=capture['capture']['casks'][0]
 assert record['desc']=='read_denied=true;write_denied=true',record
 assert sentinel.read_text()==secret,'host sentinel modified'
 assert secret not in json.dumps(capture) and secret not in r.stdout and secret not in r.stderr
 result={'result':'pass','system':system,'host_positive_control':True,'home_read_denied':True,'home_write_denied':True,'sentinel_not_disclosed':True,'description':record['desc']}
 (base/'credential-probe-summary.json').write_text(json.dumps(result,indent=2));print(json.dumps(result),flush=True)
finally:
 sentinel.unlink()
