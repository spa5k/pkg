import hashlib,json,os,subprocess,tempfile
from pathlib import Path
root=Path(tempfile.mkdtemp(prefix='pkg-native-apps-',dir='/tmp'))
app_home=root/'home'; app_home.mkdir()
state=root/'state with spaces';state.mkdir()
env=os.environ.copy();env.update({'HOME':str(app_home),'XDG_STATE_HOME':str(state),'XDG_CONFIG_HOME':str(root/'config'),'XDG_CACHE_HOME':str(root/'cache'),'PATH':'/nix/var/nix/profiles/default/bin:/usr/bin:/bin:/usr/sbin:/sbin'})
profile=state/'nix/profiles/pkg';profile.parent.mkdir(parents=True)
source=state/'pkg/taps/src/test/apps/current'; source.mkdir(parents=True)
source.joinpath('flake.nix').write_text('''{
 inputs.nixpkgs.url = "github:NixOS/nixpkgs/567a49d1913ce81ac6e9582e3553dd90a955875f";
 outputs = {nixpkgs,...}: let pkgs=nixpkgs.legacyPackages.aarch64-darwin; in {
 packages.aarch64-darwin.signed = pkgs.runCommand "pkg-assessment-signed" { raycast = builtins.storePath "/nix/store/03ys0winfk0wqssi6j981vy126as5jqz-cask-raycast-1.104.25"; } ''
 mkdir -p "$out/Applications"
 cp -a "$raycast/Applications/Raycast.app" "$out/Applications/Raycast.app"
 '';
 packages.aarch64-darwin.unsigned = pkgs.runCommand "pkg-assessment-unsigned" {} ''
 mkdir -p "$out/Applications/Unsigned.app/Contents/MacOS"
 printf '%s' '<?xml version="1.0"?><plist version="1.0"><dict><key>CFBundleExecutable</key><string>unsigned</string><key>CFBundleIdentifier</key><string>test.pkg.unsigned</string><key>CFBundleName</key><string>Unsigned</string><key>CFBundlePackageType</key><string>APPL</string></dict></plist>' > "$out/Applications/Unsigned.app/Contents/Info.plist"
 printf '#!/bin/sh\\nexit 0\\n' > "$out/Applications/Unsigned.app/Contents/MacOS/unsigned"
 chmod +x "$out/Applications/Unsigned.app/Contents/MacOS/unsigned"
 '';
 };
}''')
from urllib.parse import quote
reference='path:'+quote(str(source),safe='/')
records=[]
def run(args,expected=0):
 r=subprocess.run(args,env=env,capture_output=True,text=True,timeout=500)
 item={'args':args,'exit':r.returncode,'stdout':r.stdout[-6000:],'stderr':r.stderr[-6000:]};records.append(item);print(json.dumps(item),flush=True)
 if expected is not None: assert r.returncode==expected,item
 return r
run(['nix','profile','add','--impure','--profile',str(profile),'--no-write-lock-file',reference+'#signed'])
run(['/tmp/pkg','apps','sync'])
destination=app_home/'Applications/pkg'
def snapshot():
 items={}
 for p in sorted(destination.rglob('*')):
  if p.is_symlink(): items[str(p.relative_to(destination))]='link:'+os.readlink(p)
  elif p.is_file(): items[str(p.relative_to(destination))]=hashlib.sha256(p.read_bytes()).hexdigest()
 return items
before=snapshot();assert before and any('Raycast' in k for k in before)
run(['nix','profile','add','--impure','--profile',str(profile),'--no-write-lock-file',reference+'#unsigned'])
negative=run(['/tmp/pkg','apps','sync'],expected=None);assert negative.returncode!=0
assert 'codesign' in negative.stderr and 'Unsigned.app' in negative.stderr
assert before==snapshot(),'a rejected app changed launchers'
result={'result':'pass','root':str(root),'signed_exposed':True,'unsigned_rejected':True,'old_launchers_unchanged':True,'launcher_files':len(before),'commands':records}
(root/'evidence.json').write_text(json.dumps(result,indent=2));print(json.dumps(result),flush=True)
