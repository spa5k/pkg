import json,subprocess,sys,platform,time
from pathlib import Path
base=Path('/tmp/pkg-runtime-normal')
case='timeout-only';system='aarch64-darwin' if platform.system()=='Darwin' else 'x86_64-linux'
tree=base/'fixtures/taps'/case/'Casks';tree.mkdir(parents=True,exist_ok=True)
(tree/'pkg-timeout.rb').write_text('sleep 120\ncask "pkg-timeout" do\n version "1"\n sha256 "'+'a'*64+'"\n url "https://example.com/timeout.bin"\n binary "timeout.bin"\nend\n')
matrix=base/'public-tap-check/matrix.json';original=matrix.read_bytes();j=json.loads(original);j['source_id'][case]='pubtap-fixtures/timeout-only';matrix.write_text(json.dumps(j))
start=time.monotonic()
try:
 p=subprocess.run([sys.executable,str(base/'runtime-smoke.py'),str(base),case],capture_output=True,text=True,timeout=150)
 record=json.loads((base/f'result-{system}-{case}.json').read_text())
 result={'system':system,'exit':p.returncode,'elapsed_seconds':round(time.monotonic()-start,2),'capture_present':'capture' in record,'diagnostic_bytes':len(record['stderr'].encode()),'stderr':record['stderr']}
 assert p.returncode!=0 and 'capture' not in record,result
 assert 'every cask' in record['stderr'].lower() or 'all' in record['stderr'].lower(),result
 (base/f'deadline-{system}.json').write_text(json.dumps(result,indent=2))
 print(json.dumps(result))
finally:matrix.write_bytes(original)
