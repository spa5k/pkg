import os,pty,subprocess,sys,tempfile,json,platform
from pathlib import Path
binary=sys.argv[1];root=Path(tempfile.mkdtemp(prefix='pkg consent pty ',dir='/tmp')); records=[]
for answer,approved in [('n',False),('yes',True)]:
 case=root/answer;config=case/'config/pkg';config.mkdir(parents=True);config.joinpath('config.toml').write_text('[runtime]\nnix="/does/not/exist/pkg-test-nix"\n')
 env=os.environ.copy();env.update({'HOME':str(case/'home'),'XDG_CONFIG_HOME':str(config.parent),'XDG_STATE_HOME':str(case/'state'),'XDG_CACHE_HOME':str(case/'cache')})
 master,slave=pty.openpty()
 try:
  p=subprocess.Popen([binary,'tap','add','goreleaser/tap'],stdin=slave,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env,text=True)
  os.write(master,(answer+'\n').encode());out,err=p.communicate(timeout=15)
 finally:os.close(master);os.close(slave)
 assert p.returncode!=0
 assert 'https://github.com/goreleaser/homebrew-tap' in out and 'approval covers imports and updates of this source only' in out
 assert not (case/'state/pkg/taps/registry.json').exists()
 if not approved:
  assert 'not approved' in out and not (case/'state').exists()
 else:
  assert 'nix' in err.lower() and 'not approved' not in out
 records.append({'answer':answer,'approved':approved,'exit':p.returncode,'stdout':out,'stderr':err,'registry_created':False})
print(json.dumps({'result':'pass','system':platform.system(),'records':records}))
